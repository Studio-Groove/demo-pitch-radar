        let count = 0;

        for (let i = 0; i < channelData.length; i += step) {
          const sample = channelData[i];
          sumSquares += sample * sample;
          if (Math.abs(sample) > peakVal) peakVal = Math.abs(sample);
          count++;
        }

        // CALIBRAÇÃO EXATA DO LUFS INTEGRADO (Sem corte artificial em -5.0)

        if (label.status === "REJECTED - DYNAMICS") badgeColor = "text-rose-400 border-rose-500/30 font-bold";
        if (label.status === "APPROVED") badgeColor = "text-emerald-400 border-emerald-500/30 font-bold";

        const div = document.createElement('div');
        div.className = "text-[11px] bg-slate-950/60 p-2.5 rounded border border-slate-800";
        div.innerHTML = `
          <div class="flex justify-between items-center font-bold">
            <span class="text-slate-200">${label.name}</span>
            <span class="text-[9px] px-1.5 py-0.5 rounded border ${badgeColor}">${label.status}</span>
          </div>
          <p class="text-[10px] text-slate-400 mt-1.5 leading-snug">${label.desc}</p>
        `;
        container.appendChild(div);
      });
    }
